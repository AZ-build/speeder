    // ---- Speeder leaderboard (updated for Speeder v0.5) ----
    // Every run is its own entry, so one player can hold several spots.
    // Entry IDs start with the player's random device ID. Anyone can read.
    // v0.5 change: the "is this score believable?" check is looser, because missiles
    // and stronger guns add more kills.
    match /speeder_scores/{entryId} {
      allow read: if true;
      allow create: if validSpeeder();
      allow update: if validSpeeder() && request.resource.data.score >= resource.data.score;
      allow delete: if false;

      function validSpeeder() {
        let d = request.resource.data;
        return d.keys().hasOnly(['name', 'score', 'dist', 'kills', 'v', 'ts'])
          && d.name is string && d.name.size() >= 1 && d.name.size() <= 12
          && d.score is int && d.score >= 0 && d.score < 100000000
          && d.dist is int && d.dist >= 0
          && d.kills is int && d.kills >= 0
          && d.score <= d.dist * 8 + d.kills * 3000 + 100000
          && d.ts == request.time;
      }
    }
